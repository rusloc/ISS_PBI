CREATE MATERIALIZED VIEW portal.materialized_view_shipments_tracker
TABLESPACE pg_default
AS WITH air_sea_ports AS (
         SELECT analytical__air_sea_ports_codes.code,
            analytical__air_sea_ports_codes.type,
            analytical__air_sea_ports_codes.name,
            upper(analytical__air_sea_ports_codes.iso_country::text) AS iso_country
           FROM analytical__air_sea_ports_codes
        ), unlocode_locations AS (
         SELECT upper("UnlocodeLocations".locode::text) AS locode,
            upper("UnlocodeLocations".country) AS country,
            upper("UnlocodeLocations".iata) AS iata,
            NULLIF(COALESCE("UnlocodeLocations".name_wo_diacritics, "UnlocodeLocations".name), ''::text) AS name
           FROM portal."UnlocodeLocations"
        ), ocr_data AS (
         WITH all_related_shipments AS (
                 SELECT DISTINCT fs."ID" AS shipment_id,
                    fs.iss_domain,
                    (ocr.request_payload ->> 'parent_object_id'::text)::bigint AS source_shipment_id
                   FROM portal.ocr ocr
                     JOIN focus__shipments fs ON fs."ID" = ((ocr.request_payload ->> 'parent_object_id'::text)::bigint)
                  WHERE ocr.ocr_payload IS NOT NULL AND ocr.is_deleted = false AND (ocr.request_payload ->> 'parent_object_id'::text) IS NOT NULL
                UNION
                 SELECT DISTINCT fsr."Remote ID" AS shipment_id,
                    upper(fsr."Domain"::text) AS iss_domain,
                    fsr."Shipment ID" AS source_shipment_id
                   FROM portal.ocr ocr
                     JOIN focus__shipments fs ON fs."ID" = ((ocr.request_payload ->> 'parent_object_id'::text)::bigint)
                     JOIN focus__shipments_remote_id_info fsr ON fsr."Shipment ID" = fs."ID" AND fsr.iss_domain::text = fs.iss_domain::text
                  WHERE ocr.ocr_payload IS NOT NULL AND ocr.is_deleted = false AND fsr."Remote ID" IS NOT NULL
                UNION
                 SELECT DISTINCT fsr."Shipment ID" AS shipment_id,
                    fsr.iss_domain,
                    fs."ID" AS source_shipment_id
                   FROM portal.ocr ocr
                     JOIN focus__shipments fs ON fs."ID" = ((ocr.request_payload ->> 'parent_object_id'::text)::bigint)
                     JOIN focus__shipments_remote_id_info fsr ON fsr."Remote ID" = fs."ID" AND upper(fsr."Domain"::text) = fs.iss_domain::text
                  WHERE ocr.ocr_payload IS NOT NULL AND ocr.is_deleted = false
                ), base_data_raw AS (
                 SELECT ocr.id AS ocr_id,
                    ocr.ocr_payload,
                    fe."Equipment No" AS resolved_container_number,
                    ocr.request_payload ->> 'serial_number'::text AS serial_number,
                    (ocr.request_payload ->> 'contact_id'::text)::integer AS contact_id,
                    ocr.label,
                    bool_or(ocr.label::text = 'PKL'::text) OVER (PARTITION BY (ocr.request_payload ->> 'serial_number'::text)) AS has_pkl,
                    fs_remote."Serial No" AS remote_serial_number
                   FROM portal.ocr ocr
                     JOIN focus__shipments fs ON fs."ID" = ((ocr.request_payload ->> 'parent_object_id'::text)::bigint)
                     JOIN all_related_shipments ars ON ars.shipment_id = fs."ID" AND ars.iss_domain::text = fs.iss_domain::text
                     LEFT JOIN focus__equipment fe ON fe."Master ID" = fs."Master ID"
                     LEFT JOIN focus__shipments_remote_id_info fsr ON fsr."Shipment ID" = fs."ID" AND fsr.iss_domain::text = fs.iss_domain::text
                     LEFT JOIN focus__shipments fs_remote ON fs_remote."ID" = fsr."Remote ID"
                  WHERE ocr.ocr_payload IS NOT NULL AND ocr.is_deleted = false
                ), base_data AS (
                 SELECT t.ocr_payload,
                    t.resolved_container_number,
                    t.serial_number,
                    t.contact_id,
                    t.label,
                    t.has_pkl,
                    t.remote_serial_number
                   FROM ( SELECT b.ocr_id,
                            b.ocr_payload,
                            b.resolved_container_number,
                            b.serial_number,
                            b.contact_id,
                            b.label,
                            b.has_pkl,
                            b.remote_serial_number,
                            row_number() OVER (PARTITION BY b.ocr_id ORDER BY (b.resolved_container_number IS NULL), b.resolved_container_number, (b.remote_serial_number IS NULL), b.remote_serial_number) AS rn
                           FROM base_data_raw b) t
                  WHERE t.rn = 1
                ), filtered_items AS (
                 SELECT b.contact_id,
                    b.resolved_container_number,
                    b.serial_number,
                    b.label,
                    b.remote_serial_number,
                    jsonb_agg(elem.item ||
                        CASE
                            WHEN jsonb_typeof(b.ocr_payload -> 'batches'::text) = 'array'::text THEN jsonb_build_object('batches', COALESCE(( SELECT jsonb_agg(batch.item ORDER BY batch.ordinality) AS jsonb_agg
                               FROM jsonb_array_elements(b.ocr_payload -> 'batches'::text) WITH ORDINALITY batch(item, ordinality)
                              WHERE (batch.item ->> 'invoice_line_item_id'::text) = (elem.item ->> 'invoice_line_item_id'::text)), '[]'::jsonb))
                            ELSE '{}'::jsonb
                        END ORDER BY elem.ordinality) AS filtered_invoice_line_items
                   FROM base_data b
                     CROSS JOIN LATERAL jsonb_array_elements(
                        CASE
                            WHEN jsonb_typeof(b.ocr_payload -> 'invoice_line_item'::text) = 'array'::text THEN b.ocr_payload -> 'invoice_line_item'::text
                            ELSE '[]'::jsonb
                        END) WITH ORDINALITY elem(item, ordinality)
                  WHERE b.label::text <> 'INV'::text AND (b.label::text = 'PKL'::text OR b.has_pkl = false) AND ((elem.item ->> 'Container_number'::text) = b.resolved_container_number::text OR (elem.item ->> 'Container_number'::text) IS NULL)
                  GROUP BY b.contact_id, b.resolved_container_number, b.serial_number, b.label, b.remote_serial_number
                ), structured_ocr AS (
                 SELECT f.contact_id,
                    f.resolved_container_number,
                    f.serial_number,
                    f.label,
                    f.remote_serial_number,
                    jsonb_build_object('container_number', f.resolved_container_number, 'serial_number', f.serial_number, 'remote_serial_number', f.remote_serial_number, 'packing_list', f.filtered_invoice_line_items) AS packing_list_structure
                   FROM filtered_items f
                )
         SELECT structured_ocr.contact_id,
            structured_ocr.resolved_container_number,
            structured_ocr.serial_number,
            structured_ocr.remote_serial_number,
            structured_ocr.label,
            structured_ocr.packing_list_structure
           FROM structured_ocr
        ), base AS (
         SELECT DISTINCT ON (sc."contactId", (eq.equipment_detail ->> 'equipment_no'::text), (item.item ->> 'id'::text)) row_number() OVER (PARTITION BY sc.id ORDER BY ((item.item ->> 'id'::text)::integer)) AS row_id,
            sc.id AS cache_id,
            sc."contactId" AS contact_id,
            sc."contractBranchId" AS contract_branch_id,
            sc."issDomain" AS iss_domain,
            sc."updatedAt" AS updated_at,
            sc."createdAt" AS created_at,

            item.item ->> 'operational_status'::text AS operational_status,
            item.item ->> 'financial_status'::text AS financial_status,

            item.item ->> 'master_no'::text AS master_no,

            item.item ->> 'origin_country'::text AS origin_country,

            NULLIF(upper(
                CASE
                    WHEN length(COALESCE(item.item ->> 'origin_port'::text, ''::text)) = 3 THEN (item.item ->> 'origin_country'::text) || (item.item ->> 'origin_port'::text)
                    ELSE COALESCE(item.item ->> 'origin_port'::text, ''::text)
                END), ''::text) AS origin_port_locode,
            NULLIF(upper(
                CASE
                    WHEN length(COALESCE(item.item ->> 'destination_port'::text, ''::text)) = 3 THEN (item.item ->> 'destination_country'::text) || (item.item ->> 'destination_port'::text)
                    ELSE COALESCE(item.item ->> 'destination_port'::text, ''::text)

   FROM jsonb_array_elements(item.item -> 'equipment_details'::text) WITH ORDINALITY ed_1(value, ord)) ed
                                     JOIN ocr_data ocr_data_1 ON ocr_data_1.contact_id = sc."contactId" AND ocr_data_1.serial_number = (item.item ->> 'serial_no'::text) AND (ocr_data_1.resolved_container_number IS NOT NULL AND (eq.equipment_detail ->> 'equipment_no'::text) IS NOT NULL AND ocr_data_1.resolved_container_number::text = (eq.equipment_detail ->> 'equipment_no'::text) OR (ocr_data_1.resolved_container_number IS NULL OR (eq.equipment_detail ->> 'equipment_no'::text) IS NULL) AND ed.ord = 1)) dedup), ( SELECT jsonb_agg(DISTINCT ocr_data_1.packing_list_structure) AS jsonb_agg
                           FROM ocr_data ocr_data_1
                          WHERE ocr_data_1.contact_id = sc."contactId" AND ocr_data_1.serial_number = (item.item ->> 'serial_no'::text) AND (item.item ->> 'service'::text) ~~* '%air%'::text
                         LIMIT 1)) AS aggregated_packing_lists,
                    COALESCE(( SELECT jsonb_agg(dedup2.distinct_item) AS jsonb_agg
                           FROM ( SELECT DISTINCT ocr_data_2.packing_list_structure AS distinct_item
                                   FROM ( SELECT ed_1.value,
    ed_1.ord
   FROM jsonb_array_elements(item.item -> 'equipment_details'::text) WITH ORDINALITY ed_1(value, ord)) ed
                                     JOIN ocr_data ocr_data_2 ON ocr_data_2.remote_serial_number::text = (item.item ->> 'serial_no'::text) AND (ocr_data_2.resolved_container_number IS NOT NULL AND (eq.equipment_detail ->> 'equipment_no'::text) IS NOT NULL AND ocr_data_2.resolved_container_number::text = (eq.equipment_detail ->> 'equipment_no'::text) OR (ocr_data_2.resolved_container_number IS NULL OR (eq.equipment_detail ->> 'equipment_no'::text) IS NULL) AND ed.ord = 1)) dedup2), ( SELECT jsonb_agg(DISTINCT ocr_data_2.packing_list_structure) AS jsonb_agg
                           FROM ocr_data ocr_data_2
                          WHERE ocr_data_2.remote_serial_number::text = (item.item ->> 'serial_no'::text) AND (item.item ->> 'service'::text) ~~* '%air%'::text
                         LIMIT 1)) AS remote_aggregated_packing_lists) ocr_data ON true
             LEFT JOIN portal.materialized_view_containers mvc ON mvc."shipmentId"::text = (item.item ->> 'id'::text) AND mvc."containerId"::text = (eq.equipment_detail ->> 'equipment_no'::text)
          GROUP BY sc.id, sd."shipmentId", sd."shipmentDetails", eq.equipment_detail, fue.id, item.item, latest_status.status, ocr_data.aggregated_packing_lists, ocr_data.remote_aggregated_packing_lists, booking_status.booking_confirmed_date, delivery_order_signed_status.status_date, shipment_delivered_status.status_date, estimate_cargo_status.status_date, actual_cargo_status.status_date, pod_status.status_date, gate_out_status.status_date, empty_container_returned_status.status_date, mvc.vessel, mvc.vessel_imo, mvc."wakeoId", nw.has_wakeo, nw.is_arrived_cn, nw.calculated_arrival_cn, nw.calculated_eta_cn, nw.calculated_pta_cn
          ORDER BY sc."contactId", (eq.equipment_detail ->> 'equipment_no'::text), (item.item ->> 'id'::text), sc."updatedAt" DESC
        ), all_locodes AS (
         SELECT DISTINCT u.loc
           FROM ( SELECT base_1.port_of_discharge AS loc
                   FROM base base_1
                UNION
                 SELECT base_1.origin_port_locode
                   FROM base base_1
                UNION

                   FROM base base_1) u
          WHERE u.loc IS NOT NULL
        ), locode_name AS (
         SELECT al.loc AS locode,

                   FROM air_sea_ports asp
                  WHERE length(asp.code) = 3 AND asp.code = "right"(al.loc, 3) AND asp.iso_country = "left"(al.loc, 2)
                 LIMIT 1), ( SELECT ul.name
                   FROM unlocode_locations ul
                  WHERE ul.locode = al.loc
                 LIMIT 1), ( SELECT ul.name
                   FROM unlocode_locations ul
                  WHERE ul.country = "left"(al.loc, 2) AND ul.locode = "right"(al.loc, 3)
                 LIMIT 1), ( SELECT ul.name
                   FROM unlocode_locations ul
                  WHERE ul.country = "left"(al.loc, 2) AND ul.iata = "right"(al.loc, 3)
                 LIMIT 1)) AS name
           FROM all_locodes al
        )
